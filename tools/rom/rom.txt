# PaperOS ROM recipe: what to take out of Bigme's stock B6 firmware and what to put in.
# Format:  remove <image>:<path>        delete a directory tree from that image
#          app <image>:<dir> <apk>     install an APK as a system app at <dir>/<name>.apk
#                                       (<apk> is relative to the repo root)
# Images are the logical partitions inside `super`: system_a, system_ext_a, product_a, vendor_a.
# Kept on purpose: Bigme e-ink/system services (xSettings, XrzSettings, xMenu, StandbyApp,
# xrzResService, XRZAppManager), MediaTek SystemUI/telephony framework, WebView, mainline
# modules, package installer, DocumentsUI, DuraSpeed (battery), Clock, XDict, Calendar provider.

# --- System configuration
# USB debugging on by default (Android still asks once on screen to trust the computer).
prop vendor_a:/build.prop persist.sys.usb.config=mtp,adb
# English unless the user picks another language in PaperOS.
prop system_a:/system/build.prop persist.sys.locale=en-US
# PaperOS permissions.
file system_a:/system/etc/permissions/privapp-permissions-paperos.xml tools/rom/files/privapp-permissions-paperos.xml
